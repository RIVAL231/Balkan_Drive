## AUTHENTICATION SYSTEM

### Login Form
```graphql
mutation Login($email: String!, $password: String!) {
  login(email: $email, password: $password) {
    token
    user {
      id
      username
      email
      role
    }
  }
}
```

## FILE OPERATIONS

### Upload File
```graphql
mutation UploadFile($filename: String!, $filetype: String!, $filePath: String!, $filesize: Int!, $isPublic: Boolean!, $file: Upload!) {
  uploadFile(filename: $filename, filetype: $filetype, filePath: $filePath, filesize: $filesize, isPublic: $isPublic, file: $file) {
    uploadUrl
    uploadToken
  }
}
```

### Complete Upload
```graphql
mutation CompleteUpload($uploadToken: String!) {
  completeUpload(uploadToken: $uploadToken) {
    id
    filename
    filetype
    filesize
    isPublic
    filepath
    filehash
    createdAt
    owner {
      id
      username
    }
  }
}
```

### List Files
```graphql
query ListFiles($folder: String, $isPublic: Boolean, $ownerId: ID) {
  listFiles(folder: $folder, isPublic: $isPublic, ownerId: $ownerId) {
    id
    filename
    filetype
    filesize
    isPublic
    filepath
    filehash
    createdAt
    owner {
      id
      username
    }
    folder {
      id
      name
    }
  }
}
```
**Response Structure:**
```json
{
  "data": {
    "listFiles": [
      {
        "id": "file123",
        "filename": "document.pdf",
        "filetype": "application/pdf",
        "filesize": 2048576,
        "isPublic": false,
        "filepath": "/uploads/document.pdf",
        "filehash": "abc123...",
        "createdAt": "2025-09-16T10:30:00Z",
        "owner": {
          "id": "user123",
          "username": "johndoe"
        },
        "folder": null
      }
    ]
  }
}
```
